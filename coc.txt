        function handleSimAction(isCorrect, badgeToUnlock = null) {
            if (isCorrect) {
                addPoints(50);
                if (badgeToUnlock) unlockBadge(badgeToUnlock);
                if (state.simStep === 0) unlockBadge('first_step');
                state.simStep++;
                renderSimStep();
            }
        }

        function selectOsOption(idx) {
            state.simSelectedOs = idx;
            document.querySelectorAll('.os-item').forEach((el, i) => {
                if (i === idx) {
                    el.classList.add('bg-blue-200', 'font-bold');
                } else {
                    el.classList.remove('bg-blue-200', 'font-bold');
                }
            });
        }

        function validateOsStep() {
            if (state.simSelectedOs === null) {
                triggerMistake("Please select an operating system edition first!");
                return;
            }
            if (state.simSelectedOs === 1) {
                // Correct choice: Standard Desktop Experience
                handleSimAction(true, 'gui_master');
            } else {
                triggerMistake("In CSS NC II tasks, you MUST select 'Windows Server 2019 Standard Evaluation (Desktop Experience)' to get GUI tools!");
            }
        }

        function validateLicenseStep() {
            const checked = document.getElementById('license-check')?.checked;
            if (checked) {
                handleSimAction(true);
            } else {
                triggerMistake("You must check 'I accept the license terms' box to proceed with installation!");
            }
        }

        function createPartitionAction() {
            if (state.simPartitionCreated) return;
            
            // Simulating prompt warning
            showCustomModal(
                "Windows Setup",
                "To ensure that all Windows features work correctly, Windows might create additional partitions for system files.",
                () => {
                    state.simPartitionCreated = true;
                    addPoints(25);
                    renderSimStep();
                }
            );
        }

        function selectPartition(num) {
            if (num === 2) {
                state.simPartitionSelected = true;
                document.getElementById('part-2')?.classList.add('bg-blue-200', 'font-bold');
                document.getElementById('part-1')?.classList.remove('bg-blue-200', 'font-bold');
            } else {
                state.simPartitionSelected = false;
                triggerMistake("Do not install Windows on the System Reserved partition! Select Partition 2 (Primary).");
            }
        }

        function validatePartitionStep() {
            if (!state.simPartitionCreated) {
                triggerMistake("You must click 'New' to create system partitions before proceeding!");
                return;
            }
            if (!state.simPartitionSelected) {
                triggerMistake("Select 'Drive 0 Partition 2' (Primary partition) before clicking Next!");
                return;
            }
            unlockBadge('partition_master');
            handleSimAction(true);
        }

        function runSimulationProgress() {
            let progress = 0;
            const bar = document.getElementById('sim-progress-bar');
            const num = document.getElementById('sim-progress-num');

            const interval = setInterval(() => {
                progress += 25;
                if (bar) bar.style.width = `${progress}%`;
                if (num) num.textContent = progress;

                if (progress >= 100) {
                    clearInterval(interval);
                    setTimeout(() => {
                        state.simStep++;
                        renderSimStep();
                    }, 600);
                }
            }, 500);
        }

        function validatePasswordStep() {
            const p1 = document.getElementById('admin-pass')?.value;
            const p2 = document.getElementById('admin-pass-confirm')?.value;

            if (!p1 || !p2) {
                triggerMistake("Please enter and confirm the Administrator password!");
                return;
            }
            if (p1 !== p2) {
                triggerMistake("Passwords do not match! Please enter matching passwords.");
                return;
            }

            // Strong password regex: 8+ chars, upper, lower, digit, special char
            const strongRegex = /^(?=.*[a-z])(?=.*[A-Z])(?=.*\d)(?=.*[!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]).{8,}$/;
            if (!strongRegex.test(p1)) {
                triggerMistake("Password does not meet complexity rules! Must be 8+ chars with Uppercase, Lowercase, Number & Symbol (e.g. P@ssw0rd123).");
                return;
            }

            addPoints(100);
            handleSimAction(true);
        }

        function triggerMistake(reason) {
            deductPoints(10);
            updateBot(reason, "error");
        }

        function updateBot(msg, type = "info") {
            const speech = document.getElementById('bot-speech-text');
            const icon = document.getElementById('bot-icon');
            const alertBox = document.getElementById('feedback-alert');
            const alertText = document.getElementById('feedback-text');
            const alertIcon = document.getElementById('feedback-icon');

            if (speech) speech.textContent = `"${msg}"`;

            if (alertBox && alertText && alertIcon) {
                alertBox.classList.remove('hidden', 'bg-red-950/80', 'text-red-300', 'border-red-800', 'bg-emerald-950/80', 'text-emerald-300', 'border-emerald-800');
                
                if (type === "error") {
                    if (icon) icon.className = "fa-solid fa-robot-dead text-red-400";
                    alertBox.classList.add('bg-red-950/80', 'text-red-300', 'border', 'border-red-800');
                    alertIcon.className = "fa-solid fa-triangle-exclamation text-red-400";
                    alertText.textContent = "Mistake detected! -10 PTS";
                } else if (type === "success") {
                    if (icon) icon.className = "fa-solid fa-face-smile-beam text-emerald-400";
                    alertBox.classList.add('bg-emerald-950/80', 'text-emerald-300', 'border', 'border-emerald-800');
                    alertIcon.className = "fa-solid fa-circle-check text-emerald-400";
                    alertText.textContent = "Great job! Keep going!";
                } else {
                    if (icon) icon.className = "fa-solid fa-robot text-cyan-400";
                    alertBox.classList.add('hidden');
                }
            }
        }

        function addPoints(pts) {
            state.user.points += pts;
            updateUserHeader();
        }

        function deductPoints(pts) {
            state.user.points = Math.max(0, state.user.points - pts);
            updateUserHeader();
        }

        function unlockBadge(badgeId) {
            if (!state.user.badges.includes(badgeId)) {
                state.user.badges.push(badgeId);
                renderBadges();
            }
        }

        function updateUserHeader() {
            document.getElementById('header-user-points').textContent = state.user.points;
            document.getElementById('user-display-name').textContent = state.user.name;
            updateLeaderboard();
        }

        function resetSimulation() {
            state.simStep = 0;
            state.simSelectedOs = null;
            state.simSelectedType = null;
            state.simPartitionCreated = false;
            state.simPartitionSelected = false;
            renderSimStep();
        }

                    Submit Assessment Answers
                </button>
            </form>`;

            container.innerHTML = html;
        }

        function submitQuiz(e) {
            e.preventDefault();
            let score = 0;

            QUIZ_QUESTIONS.forEach(q => {
                const selected = document.querySelector(`input[name="q${q.id}"]:checked`);
                if (selected && parseInt(selected.value) === q.correct) {
                    score++;
                }
            });

            const percent = Math.round((score / QUIZ_QUESTIONS.length) * 100);
            state.user.quizScore = percent;

            const quizContainer = document.getElementById('quiz-container');
            const resultCard = document.getElementById('quiz-result-card');
            const icon = document.getElementById('quiz-status-icon');
            const title = document.getElementById('quiz-result-title');
            const desc = document.getElementById('quiz-result-desc');
            const certBtn = document.getElementById('view-cert-btn');

            if (quizContainer) quizContainer.classList.add('hidden');
            if (resultCard) resultCard.classList.remove('hidden');

            if (percent >= 60) {
                // PASS
                icon.className = "w-16 h-16 rounded-full mx-auto flex items-center justify-center text-3xl bg-emerald-500/20 text-emerald-400 border border-emerald-500/40";
                icon.innerHTML = '<i class="fa-solid fa-circle-check"></i>';
                title.className = "text-2xl font-bold text-emerald-400";
                title.textContent = `Passed! Score: ${percent}% (${score}/${QUIZ_QUESTIONS.length})`;
                desc.textContent = "Congratulations! You demonstrated solid competency for COC 3 Windows Server 2019 installation.";
                if (certBtn) certBtn.classList.remove('hidden');
                
                addPoints(150);
                unlockBadge('quiz_passed');
            } else {
                // FAIL
                icon.className = "w-16 h-16 rounded-full mx-auto flex items-center justify-center text-3xl bg-red-500/20 text-red-400 border border-red-500/40";
                icon.innerHTML = '<i class="fa-solid fa-circle-xmark"></i>';
                title.className = "text-2xl font-bold text-red-400";
                title.textContent = `Needs Practice. Score: ${percent}% (${score}/${QUIZ_QUESTIONS.length})`;
                desc.textContent = "You need at least 60% passing score to earn your Certificate. Review the simulation and try again!";
                if (certBtn) certBtn.classList.add('hidden');
            }
        }

        // LEADERBOARD RENDERER
        function updateLeaderboard() {
            const body = document.getElementById('leaderboard-body');
            if (!body) return;

            // Merge current user
            const list = [...state.leaderboardData];
            const existingIdx = list.findIndex(item => item.name === state.user.name);
            
            if (existingIdx >= 0) {
                list[existingIdx].points = state.user.points;
            } else {
                list.push({
                    name: state.user.name,
                    section: state.user.section,
                    simScore: state.user.points,
                    points: state.user.points
                });
            }

            list.sort((a, b) => b.points - a.points);

            body.innerHTML = list.map((item, idx) => `
                <tr class="${item.name === state.user.name ? 'bg-cyan-950/40 font-bold border-l-4 border-l-cyan-400' : 'hover:bg-slate-900/50'}">
                    <td class="py-3 px-6">
                        ${idx === 0 ? '<i class="fa-solid fa-crown text-amber-400 mr-1"></i> #1' : `#${idx + 1}`}
                    </td>
                    <td class="py-3 px-6 text-white">${item.name}</td>
                    <td class="py-3 px-6 text-slate-400 text-xs">${item.section}</td>
                    <td class="py-3 px-6 text-center text-cyan-400">${item.simScore} XP</td>
                    <td class="py-3 px-6 text-right font-bold text-amber-400">${item.points} PTS</td>
                </tr>
            `).join('');
        }

        // BADGES RENDERER
        function renderBadges() {
            const grid = document.getElementById('badges-grid');
            if (!grid) return;

            grid.innerHTML = BADGES_DEFINITIONS.map(badge => {
                const isUnlocked = state.user.badges.includes(badge.id);
                return `
                    <div class="glass-panel p-5 rounded-2xl border ${isUnlocked ? 'border-cyan-500/40 bg-cyan-950/20' : 'border-slate-800 opacity-50'} flex items-start space-x-4">
                        <div class="w-12 h-12 rounded-xl ${isUnlocked ? 'bg-cyan-500/20 ' + badge.color : 'bg-slate-800 text-slate-600'} flex items-center justify-center text-xl shrink-0">
                            <i class="fa-solid ${badge.icon}"></i>
                        </div>
                        <div>
                            <span class="text-[10px] font-bold uppercase tracking-wider ${isUnlocked ? 'text-cyan-400' : 'text-slate-500'}">
                                ${isUnlocked ? 'Unlocked' : 'Locked'}
                            </span>
                            <h4 class="font-bold text-white text-sm mt-0.5">${badge.title}</h4>
                            <p class="text-xs text-slate-400 mt-1 leading-normal">${badge.desc}</p>
                        </div>
                    </div>
                `;
            }).join('');
        }

        function switchTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.getElementById(tabId)?.classList.remove('hidden');

            document.querySelectorAll('.nav-btn').forEach(btn => {
                btn.className = "nav-btn px-4 py-2 rounded-lg text-sm font-medium transition-all text-slate-300 hover:text-white hover:bg-slate-800";
            });

            const activeBtn = document.getElementById(`nav-${tabId.replace('tab-', '')}`);
            if (activeBtn) {
                activeBtn.className = "nav-btn px-4 py-2 rounded-lg text-sm font-medium transition-all text-cyan-400 bg-cyan-950/50 border border-cyan-800/50";
            }
        }
